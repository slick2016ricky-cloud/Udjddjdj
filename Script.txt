local Players = game:GetService("Players")
local UIS = game:GetService("UserInputService")
local RunService = game:GetService("RunService")

local player = Players.LocalPlayer
local camera = workspace.CurrentCamera

-- Tuning
local FLY_SPEED = 60
local BOOST_SPEED = 45
local NORMAL_SPEED = 16
local RGB_RATE = 0.35        -- hue cycles per second
local STROBE_INTERVAL = 0.08 -- seconds per flash

local state = { fly = false, speed = false, rgb = false, strobe = false }
local strobeTimer, strobeWhite = 0, false
local flyVelocity, flyGyro

local function getChar()
	local char = player.Character
	if not char then return end
	local hum = char:FindFirstChildOfClass("Humanoid")
	local root = char:FindFirstChild("HumanoidRootPart")
	if hum and root then return char, hum, root end
end

local function setColor(color)
	local char = player.Character
	if not char then return end
	for _, part in ipairs(char:GetChildren()) do
		if part:IsA("BasePart") and part.Name ~= "HumanoidRootPart" then
			part.Color = color
		end
	end
end

local function startFly()
	local _, hum, root = getChar()
	if not root then return end
	hum.PlatformStand = true
	flyVelocity = Instance.new("BodyVelocity")
	flyVelocity.MaxForce = Vector3.new(1e5, 1e5, 1e5)
	flyVelocity.Velocity = Vector3.zero
	flyVelocity.Parent = root
	flyGyro = Instance.new("BodyGyro")
	flyGyro.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
	flyGyro.P = 1e4
	flyGyro.Parent = root
end

local function stopFly()
	if flyVelocity then flyVelocity:Destroy() flyVelocity = nil end
	if flyGyro then flyGyro:Destroy() flyGyro = nil end
	local _, hum = getChar()
	if hum then hum.PlatformStand = false end
end

UIS.InputBegan:Connect(function(input, processed)
	if processed then return end
	local k = input.KeyCode
	if k == Enum.KeyCode.F then
		state.fly = not state.fly
		if state.fly then startFly() else stopFly() end
	elseif k == Enum.KeyCode.G then
		state.speed = not state.speed
		local _, hum = getChar()
		if hum and not state.speed then hum.WalkSpeed = NORMAL_SPEED end
	elseif k == Enum.KeyCode.H then
		state.rgb = not state.rgb
		if state.rgb then state.strobe = false end
	elseif k == Enum.KeyCode.J then
		state.strobe = not state.strobe
		if state.strobe then state.rgb = false end
	end
end)

RunService.RenderStepped:Connect(function(dt)
	local _, hum, root = getChar()
	if not hum then return end

	-- Speed boost
	if state.speed then hum.WalkSpeed = BOOST_SPEED end

	-- Fly: move relative to camera with WASD, Space up, LeftShift down
	if state.fly and flyVelocity and flyGyro then
		local dir = Vector3.zero
		if UIS:IsKeyDown(Enum.KeyCode.W) then dir += camera.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.S) then dir -= camera.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.D) then dir += camera.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.A) then dir -= camera.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.Space) then dir += Vector3.yAxis end
		if UIS:IsKeyDown(Enum.KeyCode.LeftShift) then dir -= Vector3.yAxis end
		flyVelocity.Velocity = dir.Magnitude > 0 and dir.Unit * FLY_SPEED or Vector3.zero
		flyGyro.CFrame = camera.CFrame
	end

	-- RGB cycle
	if state.rgb then
		setColor(Color3.fromHSV((os.clock() * RGB_RATE) % 1, 1, 1))
	-- Black/white strobe
	elseif state.strobe then
		strobeTimer += dt
		if strobeTimer >= STROBE_INTERVAL then
			strobeTimer = 0
			strobeWhite = not strobeWhite
			setColor(strobeWhite and Color3.new(1, 1, 1) or Color3.new(0, 0, 0))
		end
	end
end)

-- Reset fly state when respawning
player.CharacterAdded:Connect(function()
	state.fly = false
	flyVelocity, flyGyro = nil, nil
end)
